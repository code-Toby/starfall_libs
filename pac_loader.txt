local HOLOGRAMS = {}
local CLIPIDS = 1
local PAC_DATA = nil
local http_count = 1

local CUSTOM_MODEL_QUEUE = {}
local CUSTOM_MATERAL_QUEUE = {}
local GLOBAL_Materials = {}
local GLOBAL_MESHES = {}

function modelPart(partData, parent)
    local part = nil
    local pos = parent:localToWorld(partData['Position'])
    local ang = parent:localToWorldAngles(partData['Angles'])
    local size = partData['Scale'] * partData['Size']
    local model = partData['Model']
    local NAME = partData['Name']
    local ID = partData['UniqueID']
    
    local Material = partData['Material']
    
    local R = partData['Color'][1] * 255
    local G = partData['Color'][2] * 255
    local B = partData['Color'][3] * 255
    
    if partData['ClassName'] == 'model' then
        R = partData['Color'][1]
        G = partData['Color'][2]
        B = partData['Color'][3] 
    end
    
    local Alpha = (partData['Alpha'] * 255)
    local Clr = Color(R, G, B, Alpha)
    
    local isFullbright = partData['NoLighting']
    local NoCulling = partData['NoCulling']
    local isInverted = partData['Invert']
    
    if string.startsWith(model, 'https://') or string.startsWith(model, 'http://') then
        part = hologram.create(pos, ang, 'models/pac/default.mdl', size)
        part:setParent(parent)
        part:setColor(Clr)
        part:setMaterial(Material)
        table.insert(CUSTOM_MODEL_QUEUE, {['Model']=model, ['part']=part})
    else
        part = hologram.create(pos, ang, model, size)
        part:setParent(parent)
        part:setColor(Clr)
        part:setMaterial(Material)
    end
    
    if isFullbright then
        part:suppressEngineLighting(true)
    end
    
    if isInverted then
        part:setCullMode(1)
    end
    
    local final_material_value = part:getMaterials()
    if Material != '' then
        final_material_value = Material
    end
    
    table.insert(HOLOGRAMS, {['NAME']=NAME,['UID']=ID, ['part']=part, ['NoCulling']=NoCulling, ['Material'] = final_material_value})

    return part
end

function clip_part(partData, parent)
    parent:setClip(CLIPIDS, true, partData['Position'], partData['Angles']:getForward(), parent)
    CLIPIDS = CLIPIDS + 1
end

function CreatePart(parts, current_part, OnFinished)
    for _, part_data in pairs(parts) do
        local holo = nil
        
        if part_data['self']['ClassName'] == 'model2' or part_data['self']['ClassName'] == 'model' then
            holo = modelPart(part_data['self'], current_part)
        end        
        
        if part_data['self']['ClassName'] == 'clip2' or part_data['self']['ClassName'] == 'clip' then
            clip_part(part_data['self'], current_part)
        end
        
        CreatePart(part_data['children'], holo) 
    end
    
    if OnFinished then
        timer.create(table.address(parts)..'_finished_dly', 0.5, 1, function()
            OnFinished()
        end)
    end
end

function load_pac_file(filename)
    if not file.exists('pac_to_holo/') then
        print('Missing pac_to_holo dir')
        print('Creating pac_to_holo dir')
        
        file.createDir('pac_to_holo')
        
        print('Directory created you can now place your pac file in: ')
        print('data/sf_filedata/pac_to_holo/')
        print('happy loading :D')
        
        return
    end
    
    if not file.exists('pac_to_holo/'..filename..'.txt')then
        print('Missing File: '..filename..'.txt')
        return
    else
        local pac_file = file.open('pac_to_holo/'..filename..'.txt', 'r')
        local raw_data = pac_file:read()
        local call, err = loadstring("return {"..raw_data.."}")
        if call == nil then print("Failed to compile! Error: " .. err) return end
        local data = call()
        
        net.start('sf_pac_netSend')
            net.writeStream(json.encode(data))
        net.send()
    end
end

function load_pacDATA(loaded_func)
    net.receive('sf_pac_netSend', function()
        net.readStream(function (cb)
            if cb == nil then 
                print('Data Returned NILL')
                return 
            else
                PAC_DATA = json.decode(cb)
                loaded_func()
            end
        end)
    end) 
end

function load_pac(group_name, ent_to_parrent_to, finished_func)
    HOLOGRAMS = {}
    for _, group in pairs(PAC_DATA) do
        if group['self']['Name'] == group_name and group['self']['ClassName'] == "group" then
            CreatePart(group['children'], ent_to_parrent_to, function()
                finished_func(HOLOGRAMS)
                handleMaterials()
                handleMeshs()          
            end)
        end
    end
    
    return HOLOGRAMS
end

function handleMeshs()
    if SERVER then
        net.start('handleMeshs')
        net.writeTable(CUSTOM_MODEL_QUEUE)
        net.send(find.allPlayers())
    end
    if CLIENT then
        net.receive('handleMeshs', function()
            CUSTOM_MODEL_QUEUE = net.readTable()
            
            local function build_mesh(url, Mesh_Data, part)
                local MeshTables, MeshData = {}
                local Mesh = nil
                
                local loading_mesh = coroutine.wrap(function() 
                    MeshTables, MeshData = mesh.parseObj(Mesh_Data, true, true)  
                    return true 
                end)
                    
                hook.add("think","loadingMesh",function()
                    while chip():getQuotaAverage()<chip():getQuotaMax()/2 do
                        if loading_mesh() then
                            MeshTables = table.clearKeys(MeshTables)
                                        
                            local finished_mesh_table = {}
                            for k, v in pairs(MeshTables) do
                                table.add(finished_mesh_table, MeshTables[k])
                            end
                                        
                            Mesh = mesh.createFromTable(finished_mesh_table, false)
                            part:setMesh(Mesh)
                            part:setRenderBounds(Vector(-200),Vector(200))
                            
                            --GLOBAL_MESHES[url] = Mesh
                                 
                            hook.remove("think","loadingMesh")
                            return
                        end
                    end
                end)
            end
            
            local function handle_url(url, part)
                local co = coroutine.running()
                if string.startsWith(url, 'https://www.dropbox.com/') then                    
                    http_count = http_count + 1
                    http.get(http.urlDropboxToRaw(url), function(body)
                        
                        build_mesh(url, body, part)
                        
                        if http_count >= http.getMaximumRequests() then
                            coroutine.resume(co)
                        end
                        
                        http_count = http_count - 1
                    end)
                elseif string.startsWith(url, 'https://drive.google.com/') then
                    http_count = http_count + 1
                    http.get(http.urlGoogleDriveToRaw(url), function(body)
                        
                        build_mesh(url, body, part)
                        
                        if http_count >= http.getMaximumRequests() then
                            coroutine.resume(co)
                        end
                        
                        http_count = http_count - 1
                    end)
                end
                
                if http_count >= http.getMaximumRequests() then
                    coroutine.yield()
                end
            end
            
            coroutine.wrap(function()
                for _, v in pairs(CUSTOM_MODEL_QUEUE) do
                    handle_url(v['Model'], v['part'])
                end
            end)()
        end)
    end
end

function handleMaterials()
    if SERVER then
        net.start('handleMaterials')
        net.writeTable(HOLOGRAMS)
        net.send(find.allPlayers())
    else
        net.receive('handleMaterials', function()
            HOLOGRAMS = net.readTable()
            
            local function MaterialLayout(mtl, url, width, height, layout)
                if not width or not height then
                    if isValid(mtl) then
                        mtl:setTexture("$basetexture", "hunter/myplastic")
                    end
                    print("Invalid texture URL")
                    return
                end
                
                local xScale = 1024 / width
                local yScale = 1024 / height
                layout(0, 0, width * xScale, height * yScale, true)
            end
            
            local function createNormalMaterial(path, part)
                if type(path) == 'table' then
                    local subID = 0
                    for k, v in pairs(path) do
                        for savedMat_Path, savedMat in pairs(GLOBAL_Materials) do
                            if savedMat_Path == v then
                                part:setSubMaterial(subID, '!'..savedMat:getName())
                                
                                subID = subID + 1
                                return
                            end
                        end
                        
                        local CustomMat = material.create('VertexLitGeneric')
                        CustomMat:setTexture('$basetexture', v)
                        CustomMat:setInt('$nocull', 8192)
                        
                        part:setSubMaterial(subID,'!'..CustomMat:getName())
                        
                        GLOBAL_Materials[v] = CustomMat
                        subID = subID + 1
                    end
                else
                    for savedMat_Path, savedMat in pairs(GLOBAL_Materials) do
                        if savedMat_Path == path then
                            part:setMaterial('!'..savedMat:getName())
                            return
                        end
                    end
                    local CustomMat = material.create('VertexLitGeneric')
                    CustomMat:setTexture('$basetexture', path)
                    CustomMat:setInt('$nocull', 8192)
                    
                    part:setMaterial('!'..CustomMat:getName())
                    
                    GLOBAL_Materials[path] = CustomMat
                end
            end
            
            local function createUrlMateral(path, part, noCull)
                for savedMat_Path, savedMat in pairs(GLOBAL_Materials) do
                    if savedMat_Path == path then
                        part:setMaterial('!'..savedMat:getName())
                        return
                    end
                end
                local CustomMat = material.create('VertexLitGeneric')
                if noCull then
                    CustomMat:setInt('$nocull', 8192)
                end
                CustomMat:setTextureURL('$basetexture', path, MaterialLayout)
                
                part:setMaterial('!'..CustomMat:getName())
                
                GLOBAL_Materials[path] = CustomMat
            end
            
            local function handle_path(path, part, noCull)
                if type(path) == 'table' then
                    if noCull then
                        createNormalMaterial(path, part)
                    end
                else
                    if string.startsWith(path, 'https://') then
                        if string.startsWith(path, 'https://i.imgur.com/') then
                            createUrlMateral(path, part, noCull)
                        end
                        if string.startsWith(path, 'https://drive.google.com/') then
                            print('sorry cant support google drive textures rn still working on a fix')
                            --createUrlMateral(http.urlGoogleDriveToRaw(path), part, noCull)
                        end
                        if string.startsWith(path, 'https://www.dropbox.com/') then
                            createUrlMateral(http.urlDropboxToRaw(path), part, noCull)
                        end
                    else
                        if noCull then
                            createNormalMaterial(path, part)
                        end
                    end
                end
            end
            
            for _, v in pairs(HOLOGRAMS) do
                handle_path(v['Material'], v['part'], v['NoCulling'])  
            end
        end) 
    end
end

function getPart(hologram_table, part_name)
    for _, v in pairs(hologram_table) do
        if v['NAME'] == part_name then
            return v['part'] 
        end
        if v['UID'] == part_name then
            return v['part']
        end
    end
end


return {['handleMeshs'] = handleMeshs, ['load_pac_file']=load_pac_file, ['load_pac']=load_pac, ['handleMaterials']=handleMaterials, ['getPart']=getPart, ['load_pacDATA']=load_pacDATA}
